#DROP DATABASE IF EXISTS Company;

CREATE DATABASE Company;

USE Company;

CREATE TABLE clients(
id_client INT AUTO_INCREMENT,
client_name VARCHAR(100) NOT NULL,
client_lastnames VARCHAR(200) NOT NULL,
client_dni VARCHAR(9) UNIQUE NOT NULL,
client_address VARCHAR(200) NOT NULL,
client_bdate DATE NOT NULL,
PRIMARY KEY(id_client)
);

CREATE TABLE providers(
id_provider INT AUTO_INCREMENT,
provider_NIF VARCHAR(9) UNIQUE NOT NULL,
provider_name VARCHAR(100) NOT NULL,
provider_address VARCHAR(200) NOT NULL,
PRIMARY KEY(id_provider)
);

CREATE TABLE products(
id_product INT AUTO_INCREMENT,
product_name VARCHAR(100) NOT NULL,
product_code VARCHAR(50) NOT NULL,
product_price VARCHAR(5) NOT NULL,
PRIMARY KEY(id_product),
fk_id_provider INT NOT NULL,
FOREIGN KEY(fk_id_provider) REFERENCES providers(id_provider)
);

CREATE TABLE purchases(
id_purchase INT AUTO_INCREMENT,
fk_id_client INT NOT NULL,
fk_id_product INT NOT NULL,
FOREIGN KEY(fk_id_client) REFERENCES clients(id_client),
FOREIGN KEY(fk_id_product) REFERENCES products(id_product),
PRIMARY KEY(id_purchase)
);
